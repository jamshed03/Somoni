title: Classes and interfaces
intro: Classes with a constructor, visibility, inheritance and interfaces.
note: public, private, protected, readonly, static, abstract
note: extends and super, override is checked
note: instanceof narrows the type
---
interface Shape {
    area(): float;
}

class Circle implements Shape {
    private r: float;

    constructor(r: float) {
        this.r = r;
    }

    area(): float {
        return 3.14 * this.r * this.r;
    }
}
